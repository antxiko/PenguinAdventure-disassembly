# Semillas a mano para el trazador de cartucho entero.
#   pNN 0xADDR b6000 b8000 bA000   # por que
#
# Solo se pone aqui lo que el trazador NO puede deducir: destinos de saltos
# indirectos cuya tabla no va pegada detras de un `call 0x4060`, y trozos de
# codigo a los que se llega cruzando la frontera de un banco.
#
# ----------------------------------------------------------------------------
# EL DESPACHADOR DEL BANCO 9 (p09:A8D0 `jp (hl)`)
#
# No es el despachador de la casa: aqui la tabla se indexa a mano y va pegada
# justo detras del salto.
#
#     A8C1  ld a,c / dec a / add a,a       el indice, por dos
#     A8C4  ld hl,0xA8D1                   la tabla, que empieza en la
#                                          instruccion siguiente al `jp (hl)`
#     A8C7  add a,l / ld l,a / jr nc / inc h
#     A8CC  ld e,(hl) / inc hl / ld d,(hl) / ex de,hl
#     A8D0  jp (hl)
#
# Son QUINCE palabras: de 0xA8D1 a 0xA8EE. Lo dice la propia tabla, no una
# suposicion -la palabra 16 vale 0x0F21, que ni siquiera es una direccion del
# cartucho, y las quince primeras caen todas entre 0xB581 y 0xBE52, dentro del
# banco 9-. El reparto de bancos es el unico con el que el trazado llega a
# 0xA8D0: 7 en 0x6000, 8 en 0x8000 y 9 en 0xA000.
p09 0xB581 7 8 9   # tabla de 0xA8D1, entrada 0
p09 0xB60A 7 8 9   # tabla de 0xA8D1, entrada 1
p09 0xB60C 7 8 9   # tabla de 0xA8D1, entrada 2
p09 0xB60E 7 8 9   # tabla de 0xA8D1, entrada 3
p09 0xB610 7 8 9   # tabla de 0xA8D1, entrada 4
p09 0xB6F4 7 8 9   # tabla de 0xA8D1, entrada 5
p09 0xB873 7 8 9   # tabla de 0xA8D1, entrada 6
p09 0xB909 7 8 9   # tabla de 0xA8D1, entrada 7
p09 0xB90B 7 8 9   # tabla de 0xA8D1, entrada 8
p09 0xBAE0 7 8 9   # tabla de 0xA8D1, entrada 9
p09 0xBB9B 7 8 9   # tabla de 0xA8D1, entrada 10
p09 0xBCBF 7 8 9   # tabla de 0xA8D1, entrada 11
p09 0xBD0F 7 8 9   # tabla de 0xA8D1, entrada 12
p09 0xBE01 7 8 9   # tabla de 0xA8D1, entrada 13
p09 0xBE52 7 8 9   # tabla de 0xA8D1, entrada 14

# ----------------------------------------------------------------------------
# EL CALLBACK DE LA SEGUNDA CABECERA (p00:40F7)
#
# No lo llama ni una instruccion del cartucho: lo llama el OTRO cartucho. La
# segunda cabecera de 0x4010 acaba en 0x4021 con los bytes F7 40, o sea 0x40F7,
# y el Konami Game Master se los guarda en su 0xD314 y salta ahi por CALSLT
# (`ld ix,(0xD314)` en su 0x5562).
#
# Que es codigo, y ademas SUYO, lo dice lo primero que hace: `ld a,(0xD31A)`,
# que es una direccion de la RAM del Game Master -donde este apunta la fase que
# ha pedido el usuario en su MODIFY MODE-. Un dato del juego no leeria ahi.
p00 0x40F7 1 2 3   # callback de la cabecera CD: el Game Master salta aqui por CALSLT con la direccion que le da 0x4021

# ----------------------------------------------------------------------------
# LA SEGUNDA TABLA DE QUINCE (p09:A98A)
#
# El banco 9 tiene DOS tablas hermanas de quince entradas, una por cada clase de
# objeto que el guion de la fase puede sacar: la de 0xA8D1 dice que rutina MONTA
# el objeto y la de 0xA98A cual lo ATIENDE en cada cuadro. Las dos se leen con
# el mismo trozo de codigo (p09:A8C7), y por eso el trazador no puede seguir
# ninguna de las dos solo: la direccion de la tabla llega en HL.
#
# Sin estas quince, la mitad del banco 9 sale como datos, y ademas de la peor
# manera: 0xB5AD, que es la rutina que atiende al primer objeto, quedaba dentro
# de un bloque de "datos" que empieza en 0xB59C -donde si hay datos de verdad,
# los quince bytes de la plantilla que 0xB58E copia al hueco-.
p09 0xB5AD 7 8 9   # tabla de 0xA98A, entrada 0
p09 0xB60B 7 8 9   # tabla de 0xA98A, entrada 1
p09 0xB60D 7 8 9   # tabla de 0xA98A, entrada 2
p09 0xB60F 7 8 9   # tabla de 0xA98A, entrada 3
p09 0xB6C8 7 8 9   # tabla de 0xA98A, entrada 4
p09 0xB71B 7 8 9   # tabla de 0xA98A, entrada 5
p09 0xB8AE 7 8 9   # tabla de 0xA98A, entrada 6
p09 0xB90A 7 8 9   # tabla de 0xA98A, entrada 7
p09 0xB9C0 7 8 9   # tabla de 0xA98A, entrada 8
p09 0xBB1E 7 8 9   # tabla de 0xA98A, entrada 9
p09 0xBC3C 7 8 9   # tabla de 0xA98A, entrada 10
p09 0xBCFB 7 8 9   # tabla de 0xA98A, entrada 11
p09 0xBD33 7 8 9   # tabla de 0xA98A, entrada 12
p09 0xBE51 7 8 9   # tabla de 0xA98A, entrada 13
p09 0xBE7E 7 8 9   # tabla de 0xA98A, entrada 14

# ----------------------------------------------------------------------------
# LO QUE LA MAQUINA DE ESTADOS METE EN LA PILA (p02:8A55)
#
# p02:800B hace `ld hl,0x8A55` y `push hl` antes de despachar, de modo que
# cuando el estado haga `ret` no vuelva a quien le llamo sino ahi. Es un `call`
# escrito del reves, y el trazador no lo sigue porque no hay ninguna
# instruccion de salto que apunte a esa direccion.
#
# Que es codigo lo dice el primer byte: 0xCD 0xC8 0x44, o sea `call 0x44C8`,
# que es la rutina del banco 0 que lee el estado de los mandos.
p02 0x8A55 1 2 3   # la maquina de estados lo mete en la PILA (0x800B) para que el estado vuelva ahi en vez de a quien le llamo

# ----------------------------------------------------------------------------
# LA DECIMOSEXTA CLASE DE OBJETO (p09:AAC9)
#
# La tabla de ATENDER de 0xA98A tiene dieciseis entradas y la de MONTAR de
# 0xA8D1 quince. La decimosexta de atender apunta a 0xAAC9, y ahi hay codigo de
# verdad: el primer byte es 0xDD 0x7E 0x1F, o sea `ld a,(ix+0x1F)`, que es
# leer un campo del hueco de objeto. Nadie salta ahi con una instruccion, asi
# que el trazador no puede llegar solo.
p09 0xAAC9 7 8 9   # entrada 16 de la tabla de atender de 0xA98A, que tiene una mas que la de montar
